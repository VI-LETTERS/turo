# Use official Nginx image as base
FROM nginx:alpine

# Copy the HTML website to the Nginx html directory
COPY . /usr/share/nginx/html/

# The following creates the app download file (placeholder - replace with actual APK when available)
RUN touch /usr/share/nginx/html/dailytask.apk

# Configure Nginx
RUN echo 'server {\n\
    listen 80;\n\
    listen [::]:80;\n\
    server_name localhost;\n\
    root /usr/share/nginx/html;\n\
    index index.html;\n\
    location / {\n\
        try_files $uri $uri/ =404;\n\
    }\n\
}' > /etc/nginx/conf.d/default.conf

# Expose port 80
EXPOSE 80

# Start Nginx when the container launches
CMD ["nginx", "-g", "daemon off;"]
